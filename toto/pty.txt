import os
import pty
import typer

app = typer.Typer()

@app.command()
def scan(image: str, working_dir: str = "."):
    cmd = ["trivy", "image", image]
    
    # Save the original directory so we can switch back later if needed
    original_cwd = os.getcwd()
    
    try:
        # Change the working directory of the Python script
        if working_dir != ".":
            os.chdir(working_dir)
            typer.echo(f"Changed working directory to: {os.getcwd()}")
            
        typer.echo(f"Target: {image}\n")
        
        # pty.spawn will now execute in the new directory
        exit_status = pty.spawn(cmd)
        
        if exit_status != 0:
            raise typer.Exit(code=exit_status)
            
    except KeyboardInterrupt:
        typer.echo("\n[Cancelled] Interrupted by user.", err=True)
        raise typer.Exit(code=130)
        
    finally:
        # Always good practice to change back to the original directory
        os.chdir(original_cwd)

if __name__ == "__main__":
    app()
